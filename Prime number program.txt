num = int(input("Enter a number: "))
count = 0
if num == 0 or num == 1:
    print("It is not a prime number")
else:
    for i in range(2,num+1):
        if num%i == 0:
            count = count+1
            
    if count > 1:
        print("It is not a prime number")
    else:
        print("It is a prime number")
