def calculate_sum(marks1,marks2,marks3):
    return marks1+marks2+marks3

#getting three inputs   
marks1 = int(input("Enter mark1: "))
marks2 = int(input("Enter mark2: "))
marks3 = int(input("Enter mark3: "))

#creating object for function
total = calculate_sum(marks1, marks2, marks3)

#Printing the total 
print("Total: ",total)
