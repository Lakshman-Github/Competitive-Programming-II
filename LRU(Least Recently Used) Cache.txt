from collections import OrderedDict

cache = OrderedDict()
capacity = 3

def put(key, value):
    if key in cache:
        cache.move_to_end(key)
        
    cache[key] = value
    
    if len(cache) > capacity:
        cache.popitem(last=False)
        
    
def get(key):
    if key in cache:
        cache.move_to_end(key)
        return cache[key]
    return -1
    
    
put(1, "A")
put(2, "B")
put(3, "C")

print(get(1))

put(4,"D")

print(get(2))
print(get(3))
print(get(4))
