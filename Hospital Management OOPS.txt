class HospitalEmployee:
    
    def __init__(self,name,employee_id):
	#default constructor python automatically calls 
        self.name = name
        self.employee_id = employee_id
        
    def work(self):
        print(self.name,"is working at the hospital")
        
class Doctor(HospitalEmployee):
    
    def work(self):
        print(self.name, "is treating patients")
        

class Nurse(HospitalEmployee):
    
    def work(self):
        print(self.name, "is taking care of patients")
        
doctor = Doctor("Dr. Arun", 101)
nurse = Nurse("Priya", 102)

doctor.work()
nurse.work()
